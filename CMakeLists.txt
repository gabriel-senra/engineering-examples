cmake_minimum_required(VERSION 3.16)
project(TapeEngineeringExamples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(tape_examples_tests tests/test_samples.cpp)
target_include_directories(tape_examples_tests PRIVATE include)
target_link_libraries(tape_examples_tests PRIVATE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(tape_examples_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME tape_examples_tests COMMAND tape_examples_tests)
set_tests_properties(tape_examples_tests PROPERTIES TIMEOUT 30)
